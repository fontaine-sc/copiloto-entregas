function mostrarEntregas(dados){ let html = ""; let extras = 0; document.getElementById("pacotes").innerText = dados.length; let contagemParadas = {}; dados.forEach(item=>{ const stop = item.Stop || item.stop || "-"; const codigo = item["SPX TN"] || item["SPX_TN"] || "-"; const endereco = item["Destination Address"] || item["Address"] || "-"; const lat = item.Latitude; const lng = item.Longitude; let titulo = ""; let classe = "stop"; if(stop === "-" || stop === "" || stop == null){ extras++; titulo = "⭐ Extra"; classe = "extra"; }else{ if(!contagemParadas[stop]){ contagemParadas[stop] = 0; } contagemParadas[stop]++; const letras = [ "", "A","B","C","D","E","F","G","H","I","J" ]; const sufixo = letras[contagemParadas[stop]] || ""; titulo = `📍 Parada ${stop}${sufixo}`; } html += `
${titulo}
${codigo}
${endereco}
`; }); document.getElementById("extras").innerText = extras; document.getElementById("lista").innerHTML = html; }
